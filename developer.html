CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "postgis";

-- 1. ENTERPRISE USERS & ROLES
CREATE TABLE erp_users (
    user_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    company_id VARCHAR(50) NOT NULL DEFAULT '22901',
    full_name VARCHAR(150) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    user_role VARCHAR(50) NOT NULL, -- ROLE_SUPER_ADMIN, ROLE_HEAD_ACCOUNTANT, etc.
    status VARCHAR(20) DEFAULT 'ACTIVE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. FARMERS & OLIVE SUPPLIERS DIRECTORY
CREATE TABLE suppliers_farmers (
    supplier_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    company_id VARCHAR(50) NOT NULL DEFAULT '22901',
    farmer_code VARCHAR(50) UNIQUE NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    village VARCHAR(100) NOT NULL,
    phone VARCHAR(30) NOT NULL,
    olive_quality_grade VARCHAR(20) DEFAULT 'EXTRA_VIRGIN',
    total_land_dunums NUMERIC(10,2) DEFAULT 0.00,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. PRODUCTS & INVENTORY BATCHES
CREATE TABLE inventory_products (
    product_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    sku VARCHAR(50) UNIQUE NOT NULL,
    product_name_en VARCHAR(150) NOT NULL,
    product_name_ar VARCHAR(150) NOT NULL,
    category VARCHAR(50) NOT NULL,
    unit_price_usd NUMERIC(10,2) NOT NULL,
    cost_usd NUMERIC(10,2) NOT NULL,
    stock_quantity NUMERIC(10,2) DEFAULT 0,
    reorder_level NUMERIC(10,2) DEFAULT 100,
    warehouse_code VARCHAR(20) DEFAULT 'WH-1'
);

-- 4. SALES ORDERS & POS TRANSACTIONS
CREATE TABLE sales_orders (
    order_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    order_number VARCHAR(50) UNIQUE NOT NULL,
    customer_name VARCHAR(150) NOT NULL,
    total_usd NUMERIC(10,2) NOT NULL,
    total_lbp NUMERIC(15,2) NOT NULL,
    exchange_rate NUMERIC(10,2) DEFAULT 89500.00,
    payment_method VARCHAR(50) DEFAULT 'CASH_USD',
    order_status VARCHAR(30) DEFAULT 'COMPLETED',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. SUPERSONIC FLEET DRIVER ROUTE DISPATCH
CREATE TABLE fleet_trips (
    trip_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    driver_name VARCHAR(100) NOT NULL,
    target_village VARCHAR(100) NOT NULL,
    packages_count INT DEFAULT 0,
    route_geometry GEOMETRY(LineString, 4326),
    trip_status VARCHAR(30) DEFAULT 'DISPATCHED',
    dispatched_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
